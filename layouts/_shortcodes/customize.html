{{- /* One customize-field token: a substitution point that shows the reader's stored value for the field (params.customize id) in place of the shipped default. The server-rendered default is real text — the markdown and print outputs carry it unchanged, so AI agents and print readers always see a concrete value. Usage: {{< customize user >}} or {{< customize cluster >}} — inside or outside a code fence. In HTML output the persistence runtime (the Customize page's form) replaces the token text with the stored value; in every other output format the default prints as-is. Tokens are alphanumerics + underscore: Chroma never splits them across highlight spans, so DOM text replacement inside highlighted code is safe (verified against Chroma's sh lexer 2026-10-30). Do not nest or use inside link destinations. */ -}} {{- $id := "" -}} {{- if .IsNamedParams -}} {{- $id = strings.TrimSpace (printf "%v" (.Get "id" | default "")) -}} {{- else -}} {{- $id = strings.TrimSpace (printf "%v" (.Get 0 | default "")) -}} {{- end -}} {{- $ok := true -}} {{- if not $id }}{{ warnf "shortcode %q requires a field id at %s; skipping the token" .Name .Position }}{{ $ok = false }}{{ end -}} {{- $field := "" -}} {{- if $ok -}} {{- $fields := $.Page.Site.Params.customize -}} {{- if not $fields -}} {{- warnf "shortcode %q: params.customize is not configured at %s; skipping the token" .Name .Position -}} {{- $ok = false -}} {{- else -}} {{- range $fields -}} {{- if eq (printf "%v" .id) $id }}{{ $field = . }}{{ end -}} {{- end -}} {{- if not $field -}} {{- warnf "shortcode %q: params.customize has no field id %q at %s; skipping the token" .Name $id .Position -}} {{- $ok = false -}} {{- end -}} {{- end -}} {{- end -}} {{- if $ok -}} {{- $format := lower ($.Page.Store.Get "tdOutputFormat" | default "html") -}} {{- if eq $format "markdown" -}} {{ $field.default }} {{- else -}} {{ $field.default }} {{- end -}} {{- end -}}
