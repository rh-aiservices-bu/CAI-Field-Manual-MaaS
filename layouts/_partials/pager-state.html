{{- /* Pinned copy of oink v1.4.0 layouts/_partials/pager-state.html. The only change is the flatten root below: params.ui.docs_sidebar_root is `home` here, so the nav tree spans the whole site and the theme would page straight out of a manual into the next top-level entry (manuals → Chat → Why this approach). Prev/next stays a reading chain inside the current top-level section instead — the manual's own sequence — which is what the pager meant before the tree was re-rooted. Re-diff on a theme bump. */ -}} {{- $page := . -}} {{- $format := lower ($page.Store.Get "tdOutputFormat" | default "html") -}} {{- $types := slice "docs" "book" "blog" -}} {{- if isset $page.Site.Params.ui "pager_types" -}} {{- $types = index $page.Site.Params.ui "pager_types" -}} {{- end -}} {{- if not (reflect.IsSlice $types) -}} {{- warnf "params.ui.pager_types must be an array of docs, book, and/or blog; got %v; using the default [docs book blog]" $types -}} {{- $types = slice "docs" "book" "blog" -}} {{- end -}} {{- $keptTypes := slice -}} {{- range $types -}} {{- if or (ne (printf "%T" .) "string") (not (in (slice "docs" "book" "blog") .)) -}} {{- warnf "invalid params.ui.pager_types value %q (allowed: docs | book | blog); dropping it" . -}} {{- else -}} {{- $keptTypes = $keptTypes | append . -}} {{- end -}} {{- end -}} {{- $types = $keptTypes -}} {{- $enabled := and (eq $format "html") (in $types $page.Type) -}} {{- if isset $page.Params "pager" -}} {{- if ne (printf "%T" $page.Params.pager) "bool" -}} {{- warnf "front matter pager must be a boolean at %s; ignoring the override" $page.Path -}} {{- else }}{{ $enabled = and (eq $format "html") $page.Params.pager }}{{ end -}} {{- end -}} {{- $prev := false -}} {{- $next := false -}} {{- if $enabled -}} {{- /* Reading chain = the page's own top-level section, not the sidebar root (Home).
